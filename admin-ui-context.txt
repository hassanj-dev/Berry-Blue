===== apps/admin/src/app/(dashboard)/error.tsx =====
"use client";

import { useRouter } from "next/navigation";
import { startTransition } from "react";

export default function DashboardError({ reset }: { reset: () => void }) {
  const router = useRouter();

  // refresh() re-fetches the server data, then reset() re-renders the page.
  function retry() {
    startTransition(() => {
      router.refresh();
      reset();
    });
  }

  return (
    <div role="alert" className="flex flex-col gap-3 rounded-2xl border border-danger p-4">
      <h1 className="text-lg font-bold text-danger">Something went wrong</h1>
      <p className="text-sm">The action could not be completed. Try again, or reload the page.</p>
      <button
        type="button"
        onClick={retry}
        className="self-start rounded-lg border border-border px-3 py-1.5 text-sm hover:border-brand"
      >
        Try again
      </button>
    </div>
  );
}

import type { Metadata } from "next";
import { redirect } from "next/navigation";
import { getAdmin } from "@/lib/auth";
import { logout } from "@/lib/auth-actions";
import { LoginForm } from "./LoginForm";

export const metadata: Metadata = { title: "Login" };

type SearchParams = Promise<Record<string, string | string[] | undefined>>;

export default async function LoginPage({ searchParams }: { searchParams: SearchParams }) {
  if (await getAdmin()) redirect("/overview");

  const { error } = await searchParams;

  return (
    <main className="mx-auto flex min-h-dvh max-w-sm flex-col justify-center gap-6 px-4">
      <h1 className="text-2xl font-bold">Admin login</h1>

      {error === "forbidden" && (
        <div
          role="alert"
          className="flex flex-col gap-2 rounded-xl border border-border bg-surface p-3 text-sm"
        >
          <p>This account is not authorized as an admin.</p>
          <form action={logout}>
            <button type="submit" className="underline">
              Sign out
            </button>
          </form>
        </div>
      )}

      <LoginForm />
    </main>
  );
}


===== apps/admin/src/app/login/LoginForm.tsx =====
"use client";

import { useActionState } from "react";
import { login, type LoginState } from "@/lib/auth-actions";

const initialState: LoginState = { error: null };

const inputClass = "rounded-lg border border-border bg-surface px-3 py-2";

export function LoginForm() {
  const [state, formAction, pending] = useActionState(login, initialState);

  return (
    <form action={formAction} className="flex flex-col gap-4">
      <label className="flex flex-col gap-1 text-sm">
        Email
        <input name="email" type="email" required autoComplete="username" className={inputClass} />
      </label>
      <label className="flex flex-col gap-1 text-sm">
        Password
        <input
          name="password"
          type="password"
          required
          autoComplete="current-password"
          className={inputClass}
        />
      </label>

      {state.error && (
        <p role="alert" className="text-sm text-danger">
          {state.error}
        </p>
      )}

      <button
        type="submit"
        disabled={pending}
        className="rounded-lg bg-brand px-4 py-2 font-semibold text-brand-foreground disabled:opacity-60"
      >
        {pending ? "Logging in..." : "Login"}
      </button>
    </form>
  );
}

"use client";

import { useSyncExternalStore } from "react";
import { Icon } from "@/components/layout/icons";
import { t } from "@/i18n";

// <html data-theme> is set before paint by lib/theme-script.ts. We only read it and change it.
function subscribe(onChange: () => void) {
  const observer = new MutationObserver(onChange);
  observer.observe(document.documentElement, { attributes: true, attributeFilter: ["data-theme"] });
  return () => observer.disconnect();
}

function getLight() {
  return document.documentElement.dataset.theme === "light";
}

export function ThemeToggle() {
  const light = useSyncExternalStore(subscribe, getLight, () => false);

  function toggle() {
    const next = light ? "dark" : "light";
    document.documentElement.dataset.theme = next;
    try {
      localStorage.setItem("theme", next);
    } catch {
      // Storage can be blocked (private mode). The choice then lasts until the page is closed.
    }
  }

  return (
    <button
      type="button"
      aria-pressed={light}
      aria-label={t("theme.label")}
      title={t("theme.title")}
      onClick={toggle}
      className="flex size-10 shrink-0 cursor-pointer items-center justify-center rounded-full text-muted transition-colors hover:text-foreground"
    >
      <Icon name={light ? "moon" : "sun"} />
    </button>
  );

===== apps/web/components/layout/MenuShell.tsx =====

"use client";

import { Menu, X } from "lucide-react";
import { useEffect, useRef, type ReactNode } from "react";

// Closes on a link tap, an outside tap or Esc.
export function MenuShell({ label, children }: { label: string; children: ReactNode }) {
  const ref = useRef<HTMLDetailsElement>(null);

  useEffect(() => {
    function onPointerDown(event: PointerEvent) {
      if (!ref.current?.contains(event.target as Node)) ref.current?.removeAttribute("open");
    }
    function onKeyDown(event: KeyboardEvent) {
      if (event.key === "Escape") ref.current?.removeAttribute("open");
    }
    document.addEventListener("pointerdown", onPointerDown);
    document.addEventListener("keydown", onKeyDown);
    return () => {
      document.removeEventListener("pointerdown", onPointerDown);
      document.removeEventListener("keydown", onKeyDown);
    };
  }, []);

  return (
    <details ref={ref} className="group order-1 lg:hidden">
      <summary
        aria-label={label}
        className="flex size-10 cursor-pointer list-none items-center justify-center rounded-full hover:bg-foreground/10 [&::-webkit-details-marker]:hidden"
      >
        <Menu size={22} aria-hidden="true" className="group-open:hidden" />
        <X size={22} aria-hidden="true" className="hidden group-open:block" />
      </summary>
      <div
        onClick={(event) => {
          if ((event.target as HTMLElement).closest("a")) ref.current?.removeAttribute("open");
        }}
        className="absolute inset-x-3 top-full z-50 mt-2 max-h-[70dvh] overflow-y-auto rounded-2xl border border-border bg-surface p-3 shadow-xl"
      >
        {children}
      </div>
    </details>
  );
}

  const params = request.nextUrl.searchParams;
  const requested = Number(params.get("size"));
  const size = SIZES.has(requested) ? requested : 512;

  return renderIcon(size, params.get("maskable") === "1", {
    "Cache-Control": "public, max-age=86400, stale-while-revalidate=604800",
  });
}


===== apps/web/lib/pwa-icon.tsx =====

import { ImageResponse } from "next/og";
import { siteConfig } from "@/lib/site";

// The mascot from components/shared/Mascot.tsx with fixed colours (no CSS variables here).
// Maskable icons get cropped by the phone, so the picture stays in the middle 60%.
export function renderIcon(size: number, maskable: boolean, headers: Record<string, string> = {}) {
  const face = Math.round(size * (maskable ? 0.6 : 0.74));

  return new ImageResponse(
    <div
      style={{
        width: "100%",
        height: "100%",
        display: "flex",
        alignItems: "center",
        justifyContent: "center",
        background: siteConfig.themeColor,
      }}
    >
      <svg xmlns="http://www.w3.org/2000/svg" width={face} height={face} viewBox="0 0 100 100">
        <circle cx="50" cy="55" r="38" fill="#ffffff" />
        <circle cx="38" cy="50" r="6" fill={siteConfig.themeColor} />
        <circle cx="62" cy="50" r="6" fill={siteConfig.themeColor} />
        <path
          d="M38 68 Q50 78 62 68"
          stroke={siteConfig.themeColor}
          strokeWidth="4"
          fill="none"
          strokeLinecap="round"
        />
        <path d="M50 17 Q56 6 66 10 Q60 20 50 17Z" fill={siteConfig.accentColor} />
      </svg>
    </div>,
    { width: size, height: size, headers },
  );
}
