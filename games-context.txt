===== apps\web\app\(public)\games\page.tsx =====
import type { Metadata } from "next";
import { notFound } from "next/navigation";
import { Suspense } from "react";
import { GameCard, GameCardSkeleton } from "@/components/game/GameCard";
import { GamesFilterBar } from "@/components/game/GamesFilterBar";
import { Pagination } from "@/components/game/Pagination";
import { SortTabs } from "@/components/game/SortTabs";
import { t } from "@/i18n";
import { getCategories, getGamesPage } from "@/lib/games";
import { PAGE_SIZE, parseGamesQuery, type GamesFilters } from "@/lib/games-query";
import { getTags } from "@/lib/tags";

export const metadata: Metadata = {
  title: t("games.title"),
};

type SearchParams = Promise<Record<string, string | string[] | undefined>>;

export default function GamesPage({ searchParams }: { searchParams: SearchParams }) {
  return (
    <main>
      <h1 className="text-2xl font-bold md:text-4xl">{t("games.title")}</h1>
      <div className="mt-4">
        <Suspense fallback={<GamesGridSkeleton />}>
          <GamesContent searchParams={searchParams} />
        </Suspense>
      </div>
    </main>
  );
}

async function GamesContent({ searchParams }: { searchParams: SearchParams }) {
  const query = parseGamesQuery(await searchParams);
  const [categories, tags] = await Promise.all([getCategories(), getTags()]);

  // Only values that really exist reach the database (and the cache).
  const filters: GamesFilters = {
    category: categories.some((c) => c.slug === query.category) ? query.category : undefined,
    device: query.device,
    tag: tags.some((option) => option.tag === query.tag) ? query.tag : undefined,
  };
  const hasFilters = Boolean(filters.category || filters.device || filters.tag);

  const { sort, page } = query;
  const result = await getGamesPage(sort, page, filters);

  return (
    <div className="flex flex-col gap-4">
      <GamesFilterBar sort={sort} filters={filters} categories={categories} tags={tags} />
      <ResultsBody
        sort={sort}
        page={page}
        filters={filters}
        hasFilters={hasFilters}
        result={result}
      />
    </div>
  );
}

function ResultsBody({
  sort,
  page,
  filters,
  hasFilters,
  result,
}: {
  sort: Parameters<typeof SortTabs>[0]["current"];
  page: number;
  filters: GamesFilters;
  hasFilters: boolean;
  result: Awaited<ReturnType<typeof getGamesPage>>;
}) {
  if (!result) return <p className="text-muted">{t("games.error")}</p>;
  if (page > result.pageCount) notFound();

  const from = (page - 1) * PAGE_SIZE + 1;
  const to = from + result.games.length - 1;

  return (
    <>
      <SortTabs current={sort} filters={filters} />

      {result.total === 0 ? (
        <p className="text-muted">{hasFilters ? t("games.noMatch") : t("games.empty")}</p>
      ) : (
        <>
          <p className="text-sm text-muted">
            {t("games.showing", { from, to, total: result.total })}
          </p>
          <ul className="grid grid-cols-2 gap-3 md:grid-cols-4">
            {result.games.map((game, index) => (
              <li key={game.id}>
                <GameCard game={game} priority={index < 4} />
              </li>
            ))}
          </ul>
          <Pagination sort={sort} filters={filters} page={page} pageCount={result.pageCount} />
        </>
      )}
    </>
  );
}

function GamesGridSkeleton() {
  return (
    <div aria-hidden="true" className="grid grid-cols-2 gap-3 md:grid-cols-4">
      {Array.from({ length: PAGE_SIZE }, (_, i) => (
        <GameCardSkeleton key={i} />
      ))}
    </div>
  );
}

===== apps\web\components\game\GamesFilterBar.tsx =====
import Link from "next/link";
import type { Category } from "@berryblue/shared";
import { t } from "@/i18n";
import { cn } from "@/lib/cn";
import { DEVICES, gamesHref, type GamesFilters, type SortName } from "@/lib/games-query";
import type { TagOption } from "@/lib/tags";

const TAG_CHIP_LIMIT = 12;

type GamesFilterBarProps = {
  sort: SortName;
  filters: GamesFilters;
  categories: Category[];
  tags: TagOption[];
};

function Chip({
  href,
  active,
  children,
}: {
  href: string;
  active: boolean;
  children: React.ReactNode;
}) {
  return (
    <li>
      <Link
        href={href}
        aria-current={active ? "true" : undefined}
        className={cn(
          "inline-flex min-h-10 items-center rounded-full border px-4 text-sm font-medium capitalize",
          active
            ? "border-brand bg-brand text-brand-foreground"
            : "border-border bg-surface text-muted hover:border-brand",
        )}
      >
        {children}
      </Link>
    </li>
  );
}

function Group({ label, children }: { label: string; children: React.ReactNode }) {
  return (
    <div>
      <p className="mb-1 text-xs font-semibold uppercase text-muted">{label}</p>
      <nav aria-label={label} className="-mx-4 overflow-x-auto px-4 md:mx-0 md:px-0">
        <ul className="flex gap-2 whitespace-nowrap">{children}</ul>
      </nav>
    </div>
  );
}

// Plain links, no JavaScript. A chip that is already on turns that filter off.
// A new filter never keeps the old page number, so it starts again at page 1.
export function GamesFilterBar({ sort, filters, categories, tags }: GamesFilterBarProps) {
  const hasFilters = Boolean(filters.category || filters.device || filters.tag);

  const shownTags = tags.slice(0, TAG_CHIP_LIMIT).map((option) => option.tag);
  if (filters.tag && !shownTags.includes(filters.tag)) shownTags.push(filters.tag);

  return (
    <section aria-label={t("games.filters.label")} className="flex flex-col gap-3">
      {categories.length > 0 ? (
        <Group label={t("games.filters.category")}>
          <Chip
            href={gamesHref({ ...filters, sort, category: undefined })}
            active={!filters.category}
          >
            {t("games.filters.all")}
          </Chip>
          {categories.map((category) => {
            const active = filters.category === category.slug;
            return (
              <Chip
                key={category.id}
                href={gamesHref({ ...filters, sort, category: active ? undefined : category.slug })}
                active={active}
              >
                {category.name}
              </Chip>
            );
          })}
        </Group>
      ) : null}

      <Group label={t("games.filters.device")}>
        <Chip href={gamesHref({ ...filters, sort, device: undefined })} active={!filters.device}>
          {t("games.filters.anyDevice")}
        </Chip>
        {DEVICES.map((device) => {
          const active = filters.device === device;
          return (
            <Chip
              key={device}
              href={gamesHref({ ...filters, sort, device: active ? undefined : device })}
              active={active}
            >
              {t(`devices.${device}`)}
            </Chip>
          );
        })}
      </Group>

      {shownTags.length > 0 ? (
        <Group label={t("games.filters.tag")}>
          {shownTags.map((tag) => {
            const active = filters.tag === tag;
            return (
              <Chip
                key={tag}
                href={gamesHref({ ...filters, sort, tag: active ? undefined : tag })}
                active={active}
              >
                {tag.replace(/-/g, " ")}
              </Chip>
            );
          })}
        </Group>
      ) : null}

      {hasFilters ? (
        <p>
          <Link href={gamesHref({ sort })} className="text-sm text-brand hover:underline">
            {t("games.filters.clear")}
          </Link>
        </p>
      ) : null}
    </section>
  );
}

===== apps\web\components\game\Pagination.tsx =====
import Link from "next/link";
import { t } from "@/i18n";
import { cn } from "@/lib/cn";
import { gamesHref, type GamesFilters, type SortName } from "@/lib/games-query";

type PaginationProps = {
  sort: SortName;
  filters: GamesFilters;
  page: number;
  pageCount: number;
};

// Pehla, aakhri, aur maujooda page ke aas paas ke pages; beech ka farq "gap" se dikhate hain.
function pageItems(current: number, count: number): Array<number | "gap"> {
  const wanted = new Set([1, count, current - 1, current, current + 1]);
  const sorted = [...wanted].filter((p) => p >= 1 && p <= count).sort((a, b) => a - b);

  const items: Array<number | "gap"> = [];
  sorted.forEach((p, i) => {
    if (i > 0 && p - sorted[i - 1] > 1) items.push("gap");
    items.push(p);
  });
  return items;
}

const base =
  "inline-flex min-h-11 min-w-11 items-center justify-center rounded-xl border border-border px-3 text-sm";

export function Pagination({ sort, filters, page, pageCount }: PaginationProps) {
  if (pageCount <= 1) return null;

  const hrefFor = (target: number) => gamesHref({ ...filters, sort, page: target });

  return (
    <nav aria-label={t("games.pagination.label")} className="mt-8">
      <ul className="flex flex-wrap items-center justify-center gap-2">
        <li>
          {page > 1 ? (
            <Link href={hrefFor(page - 1)} rel="prev" className={cn(base, "bg-surface")}>
              {t("games.pagination.previous")}
            </Link>
          ) : (
            <span aria-disabled="true" className={cn(base, "text-muted opacity-50")}>
              {t("games.pagination.previous")}
            </span>
          )}
        </li>

        {pageItems(page, pageCount).map((item, i) =>
          item === "gap" ? (
            <li key={"gap-" + i} aria-hidden="true" className="px-1 text-muted">
              &hellip;
            </li>
          ) : (
            <li key={item}>
              <Link
                href={hrefFor(item)}
                aria-current={item === page ? "page" : undefined}
                aria-label={t("games.pagination.page", { page: item })}
                className={cn(
                  base,
                  item === page ? "bg-brand text-brand-foreground" : "bg-surface",
                )}
              >
                {item}
              </Link>
            </li>
          ),
        )}

        <li>
          {page < pageCount ? (
            <Link href={hrefFor(page + 1)} rel="next" className={cn(base, "bg-surface")}>
              {t("games.pagination.next")}
            </Link>
          ) : (
            <span aria-disabled="true" className={cn(base, "text-muted opacity-50")}>
              {t("games.pagination.next")}
            </span>
          )}
        </li>
      </ul>
    </nav>
  );
}

"use client";

import Link from "next/link";
import { usePathname } from "next/navigation";
import type { ReactNode } from "react";
import { cn } from "@/lib/cn";

type NavLinkProps = {
  href: string;
  className?: string;
  activeClassName?: string;
  children: ReactNode;
};

export function NavLink({ href, className, activeClassName, children }: NavLinkProps) {
  const pathname = usePathname();
  const active = href === "/" ? pathname === "/" : pathname.startsWith(href);

  return (
    <Link
      href={href}
      aria-current={active ? "page" : undefined}
      className={cn(className, active && activeClassName)}
    >
      {children}
    </Link>
  );
}

import { deviceSchema, type Device } from "@berryblue/shared";
import { z } from "zod";

export const SORTS = ["trending", "new", "popular"] as const;
export type SortName = (typeof SORTS)[number];

export const DEVICES = deviceSchema.options;

export const PAGE_SIZE = 12;

export type GamesFilters = {
  category?: string | undefined;
  device?: Device | undefined;
  tag?: string | undefined;
};

// The URL is not trusted: a wrong value becomes the default (or no filter).
const slugParam = z
  .string()
  .regex(/^[a-z0-9-]+$/)
  .max(60)
  .optional()
  .catch(undefined);

const querySchema = z.object({
  sort: z.enum(SORTS).catch("popular"),
  page: z.coerce.number().int().min(1).catch(1),
  category: slugParam,
  tag: slugParam,
  device: deviceSchema.optional().catch(undefined),
});

export function parseGamesQuery(params: Record<string, string | string[] | undefined>) {
  return querySchema.parse({
    sort: params.sort,
    page: params.page,
    category: params.category,
    tag: params.tag,
    device: params.device,
  });
}

// Every link on /games is built here, so sort, filters and page always travel together.
export function gamesHref(state: GamesFilters & { sort: SortName; page?: number }) {
  const params = new URLSearchParams();
  params.set("sort", state.sort);
  if (state.category) params.set("category", state.category);
  if (state.device) params.set("device", state.device);
  if (state.tag) params.set("tag", state.tag);
  if (state.page && state.page > 1) params.set("page", String(state.page));
  return "/games?" + params.toString();
}
