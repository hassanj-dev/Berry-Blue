import type { Metadata } from "next";
import { requireAdmin } from "@/lib/auth";
import { createSecretClient } from "@/lib/supabase/secret";

export const metadata: Metadata = { title: "Audit log" };

type LogRow = {
  id: string;
  admin_id: string | null;
  action: string;
  entity_type: string;
  entity_id: string | null;
  details: Record<string, unknown>;
  created_at: string;
};

export default async function AuditLogsPage() {
  await requireAdmin();
  const supabase = createSecretClient();

  const { data, error } = await supabase
    .from("audit_logs")
    .select("id, admin_id, action, entity_type, entity_id, details, created_at")
    .order("created_at", { ascending: false })
    .limit(100);

  if (error) throw new Error(`audit log failed: ${error.message}`);
  const logs = (data ?? []) as LogRow[];

  // Show admin emails instead of raw ids. Only look up the admins that appear in this log.
  const adminIds = [...new Set(logs.map((l) => l.admin_id).filter((id): id is string => !!id))];
  const emails = new Map<string, string>();
  await Promise.all(
    adminIds.map(async (id) => {
      const { data: result } = await supabase.auth.admin.getUserById(id);
      if (result.user?.email) emails.set(id, result.user.email);
    }),
  );

  return (
    <div className="flex flex-col gap-6">
      <h1 className="text-2xl font-bold">Audit log</h1>

      {logs.length === 0 ? (
        <p className="text-muted">Nothing recorded yet.</p>
      ) : (
        <div className="overflow-x-auto rounded-2xl border border-border">
          <table className="w-full text-left text-sm">
            <thead className="bg-surface text-muted">
              <tr>
                <th className="px-3 py-2">When</th>
                <th className="px-3 py-2">Admin</th>
                <th className="px-3 py-2">Action</th>
                <th className="px-3 py-2">Details</th>
              </tr>
            </thead>
            <tbody>
              {logs.map((log) => (
                <tr key={log.id} className="border-t border-border align-top">
                  <td className="whitespace-nowrap px-3 py-2">
                    {new Date(log.created_at).toLocaleString("en-GB")}
                  </td>
                  <td className="px-3 py-2">
                    {log.admin_id ? (emails.get(log.admin_id) ?? log.admin_id) : "-"}
                  </td>
                  <td className="whitespace-nowrap px-3 py-2 font-medium">{log.action}</td>
                  <td className="break-all px-3 py-2 text-muted">{JSON.stringify(log.details)}</td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      )}
    </div>
  );
}

"use client";

import { useRouter } from "next/navigation";
import { startTransition } from "react";

export default function DashboardError({ reset }: { reset: () => void }) {
  const router = useRouter();

  // refresh() re-fetches the server data, then reset() re-renders the page.
  function retry() {
    startTransition(() => {
      router.refresh();
      reset();
    });
  }

  return (
    <div role="alert" className="flex flex-col gap-3 rounded-2xl border border-danger p-4">
      <h1 className="text-lg font-bold text-danger">Something went wrong</h1>
      <p className="text-sm">The action could not be completed. Try again, or reload the page.</p>
      <button
        type="button"
        onClick={retry}
        className="self-start rounded-lg border border-border px-3 py-1.5 text-sm hover:border-brand"
      >
        Try again
      </button>
    </div>
  );
}

import type { Metadata } from "next";
import Link from "next/link";
import { notFound } from "next/navigation";
import { z } from "zod";
import { requireAdmin } from "@/lib/auth";
import { formatValidationErrors, getStaged, listCategories } from "@/lib/moderation";
import { ReviewForm } from "./ReviewForm";

export const metadata: Metadata = { title: "Review game" };

const isHttpUrl = (value: string | null): value is string => !!value && /^https?:\/\//i.test(value);

export default async function ReviewPage({ params }: { params: Promise<{ id: string }> }) {
  await requireAdmin();

  const { id } = await params;
  if (!z.uuid().safeParse(id).success) notFound();

  const game = await getStaged(id);
  if (!game) notFound();

  const errors = formatValidationErrors(game.validation_errors);
  const categories = game.status === "pending" ? await listCategories() : [];

  // Pre-select a category if the provider's category matches one of ours (admin can still change it).
  const providerCategory = game.provider_category?.trim().toLowerCase();
  const suggested =
    categories.find((c) => c.id === game.category_id) ??
    categories.find(
      (c) => c.slug === providerCategory || c.name.toLowerCase() === providerCategory,
    );

  const rows: [string, string][] = [
    ["Provider", game.provider],
    ["Provider game ID", game.provider_game_id],
    ["Provider category", game.provider_category ?? "-"],
    ["Slug", game.slug ?? "-"],
    ["Tags", game.tags.length ? game.tags.join(", ") : "-"],
    ["Devices", game.devices.length ? game.devices.join(", ") : "-"],
    ["Status", game.status],
  ];

  return (
    <div className="flex flex-col gap-6">
      <Link
        href={"/moderation?status=" + game.status}
        className="text-sm text-brand hover:underline"
      >
        &larr; Back to list
      </Link>

      <h1 className="text-2xl font-bold">{game.title ?? "(no title)"}</h1>

      {errors.length > 0 && (
        <div role="alert" className="rounded-2xl border border-danger p-4">
          <h2 className="font-bold text-danger">Validation errors</h2>
          <ul className="mt-2 list-disc pl-5 text-sm">
            {errors.map((e, i) => (
              <li key={i}>{e}</li>
            ))}
          </ul>
        </div>
      )}

      <div className="grid gap-6 md:grid-cols-[16rem_1fr]">
        <div className="flex flex-col gap-3">
          {isHttpUrl(game.thumbnail_url) ? (
            // eslint-disable-next-line @next/next/no-img-element
            <img
              src={game.thumbnail_url}
              alt=""
              className="w-full rounded-2xl border border-border"
            />
          ) : (
            <p className="rounded-2xl border border-border bg-surface p-4 text-sm text-muted">
              Thumbnail: {game.thumbnail_url ?? "none"}
            </p>
          )}
          {isHttpUrl(game.embed_url) ? (
            <a
              href={game.embed_url}
              target="_blank"
              rel="noopener noreferrer"
              className="text-sm text-brand hover:underline"
            >
              Open game in a new tab
            </a>
          ) : (
            <p className="text-sm text-muted">Embed: {game.embed_url ?? "none"}</p>
          )}
        </div>

        <div className="flex flex-col gap-4">
          <p className="text-sm">{game.description || "No description."}</p>
          <dl className="grid grid-cols-[10rem_1fr] gap-x-4 gap-y-2 text-sm">
            {rows.map(([label, value]) => (
              <div key={label} className="contents">
                <dt className="text-muted">{label}</dt>
                <dd className="break-all">{value}</dd>
              </div>
            ))}
          </dl>
        </div>
      </div>

      {game.status === "pending" ? (
        <ReviewForm
          stagedId={game.id}
          defaultSlug={game.slug ?? ""}
          defaultCategoryId={suggested?.id ?? ""}
          categories={categories.map((c) => ({ id: c.id, name: c.name }))}
        />
      ) : (
        <p className="rounded-2xl border border-border bg-surface p-4 text-sm">
          This game is <strong>{game.status}</strong>, so it cannot be approved or rejected from
          here.
        </p>
      )}
    </div>
  );
}

"use client";

import { startTransition, useActionState, useState, type FormEvent } from "react";
import { approveGame, rejectGame, type ReviewState } from "@/lib/moderation-actions";

type ReviewFormProps = {
  stagedId: string;
  defaultSlug: string;
  defaultCategoryId: string;
  categories: { id: string; name: string }[];
};

const initialState: ReviewState = { error: null };
const inputClass = "rounded-lg border border-border bg-surface px-3 py-2";

// Submitting through onSubmit (not the form's action prop) stops React 19 from clearing the fields after an error.
function submit(action: (data: FormData) => void) {
  return (event: FormEvent<HTMLFormElement>) => {
    event.preventDefault();
    const data = new FormData(event.currentTarget);
    startTransition(() => action(data));
  };
}

export function ReviewForm({
  stagedId,
  defaultSlug,
  defaultCategoryId,
  categories,
}: ReviewFormProps) {
  const [categoryId, setCategoryId] = useState(defaultCategoryId);
  const [approveState, approveAction, approving] = useActionState(approveGame, initialState);
  const [rejectState, rejectAction, rejecting] = useActionState(rejectGame, initialState);
  const busy = approving || rejecting;

  return (
    <div className="flex flex-col gap-6">
      <form
        onSubmit={submit(approveAction)}
        className="flex flex-col gap-4 rounded-2xl border border-border p-4"
      >
        <h2 className="text-lg font-bold">Approve and publish</h2>
        <input type="hidden" name="stagedId" value={stagedId} />

        <label className="flex flex-col gap-1 text-sm">
          Category
          <select
            name="categoryId"
            value={categoryId}
            onChange={(e) => setCategoryId(e.target.value)}
            className={inputClass}
          >
            <option value="">Choose a category...</option>
            {categories.map((c) => (
              <option key={c.id} value={c.id}>
                {c.name}
              </option>
            ))}
          </select>
        </label>

        <label className="flex flex-col gap-1 text-sm">
          Slug (the game URL)
          <input name="slug" defaultValue={defaultSlug} required className={inputClass} />
        </label>

        {approveState.error && (
          <p role="alert" className="text-sm text-danger">
            {approveState.error}
          </p>
        )}

        <button
          type="submit"
          disabled={!categoryId || busy}
          className="rounded-lg bg-brand px-4 py-2 font-semibold text-brand-foreground disabled:opacity-50"
        >
          {approving ? "Publishing..." : "Approve and publish"}
        </button>
        {!categoryId && (
          <p className="text-sm text-muted">Choose a category to enable this button.</p>
        )}
      </form>

      <form
        onSubmit={submit(rejectAction)}
        className="flex flex-col gap-4 rounded-2xl border border-border p-4"
      >
        <h2 className="text-lg font-bold">Reject</h2>
        <input type="hidden" name="stagedId" value={stagedId} />

        <label className="flex flex-col gap-1 text-sm">
          Reason (optional)
          <input name="reason" maxLength={500} className={inputClass} />
        </label>

        {rejectState.error && (
          <p role="alert" className="text-sm text-danger">
            {rejectState.error}
          </p>
        )}

        <button
          type="submit"
          disabled={busy}
          className="rounded-lg border border-danger px-4 py-2 font-semibold text-danger disabled:opacity-50"
        >
          {rejecting ? "Rejecting..." : "Reject"}
        </button>
      </form>
    </div>
  );
}

import type { Metadata } from "next";

export const metadata: Metadata = { title: "Moderation" };

type SearchParams = Promise<Record<string, string | string[] | undefined>>;

// Bad values in the URL fall back to "pending" instead of crashing.
const statusSchema = z.enum(STATUSES).catch("pending");

const TAB_LABELS = {
  pending: "Pending",
  invalid: "Invalid",
  rejected: "Rejected",
  approved: "Approved",

  const games = await listStaged(status);

  return (
    <div className="flex flex-col gap-6">
      <h1 className="text-2xl font-bold">Moderation</h1>

      <nav aria-label="Filter by status">
        <ul className="flex flex-wrap gap-2">
          {STATUSES.map((s) => (
            <li key={s}>
              <Link
                href={"/moderation?status=" + s}
                aria-current={s === status ? "true" : undefined}
                className={
                  "inline-flex min-h-10 items-center rounded-xl border border-border px-4 text-sm font-medium " +
                  (s === status ? "bg-brand text-brand-foreground" : "bg-surface")
                }

                {TAB_LABELS[s]}
              </Link>
            </li>
          ))}
        </ul>
      </nav>

      {games.length === 0 ? (
        <p className="text-muted">No {TAB_LABELS[status].toLowerCase()} games.</p>
      ) : (
        <ul className="flex flex-col gap-3">

                >
                  <span className="font-semibold">{game.title ?? "(no title)"}</span>
                  <span className="text-sm text-muted">

                    {new Date(game.created_at).toLocaleDateString("en-GB")}

                  {errors.length > 0 && (
                    <ul className="mt-1 list-disc pl-5 text-sm text-danger">

import type { Metadata } from "next";
import { requireAdmin } from "@/lib/auth";
import { createSecretClient } from "@/lib/supabase/secret";

export const metadata: Metadata = { title: "Overview" };

async function countStaged(status: string) {
  const { count, error } = await createSecretClient()
    .from("staged_games")
    .select("id", { count: "exact", head: true })
    .eq("status", status);

  return error ? null : count;
}

async function countLiveGames() {
  const { count, error } = await createSecretClient()
    .from("games")
    .select("id", { count: "exact", head: true })
    .eq("is_published", true);

  return error ? null : count;
}

export default async function OverviewPage() {
  await requireAdmin();

  const [pending, invalid, rejected, live] = await Promise.all([
    countStaged("pending"),
    countStaged("invalid"),
    countStaged("rejected"),
    countLiveGames(),
  ]);

  const cards = [
    { label: "Pending review", value: pending },
    { label: "Invalid", value: invalid },
    { label: "Rejected", value: rejected },
    { label: "Live games", value: live },
  ];

  return (
    <div className="flex flex-col gap-6">
      <h1 className="text-2xl font-bold">Overview</h1>
      <ul className="grid grid-cols-2 gap-3 md:grid-cols-4">
        {cards.map((card) => (
          <li key={card.label} className="rounded-2xl border border-border bg-surface p-4">

            <p className="mt-1 text-3xl font-bold">{card.value ?? "-"}</p>

export async function isAdminUser(userId: string): Promise<boolean> {
  const { data, error } = await createSecretClient()
    .from("admin_users")
    .select("user_id")
    .eq("user_id", userId)
    .maybeSingle();

  if (error) throw new Error(`admin check failed: ${error.message}`);

// getUser() verifies the token with Supabase Auth (never trust the cookie alone).
const getAuth = cache(async () => {
  const supabase = await createClient();
  const {
    data: { user },
  } = await supabase.auth.getUser();

  if (!user) return { user: null, isAdmin: false };
  return { user, isAdmin: await isAdminUser(user.id) };
});

export async function getAdmin(): Promise<Admin | null> {
  const { user, isAdmin } = await getAuth();
  return user && isAdmin ? { id: user.id, email: user.email ?? "" } : null;
}

// First line of every page and Server Action: do not rely on the layout (it does not re-run on every navigation).
export async function requireAdmin(): Promise<Admin> {
  const { user, isAdmin } = await getAuth();

  if (!user) redirect("/login");
  if (!isAdmin) redirect("/login?error=forbidden");

  return { id: user.id, email: user.email ?? "" };
}

export async function login(_previous: LoginState, formData: FormData): Promise<LoginState> {
  const parsed = loginSchema.safeParse({
    email: formData.get("email"),
    password: formData.get("password"),
  });
  if (!parsed.success) return { error: LOGIN_FAILED };

  const supabase = await createClient();
  const { data, error } = await supabase.auth.signInWithPassword(parsed.data);
  if (error || !data.user) return { error: LOGIN_FAILED };

  // Check with this user id instead of re-reading the cookie: the cookie just set may not be visible in this same request.
  if (!(await isAdminUser(data.user.id))) {
    await supabase.auth.signOut();
    return { error: LOGIN_FAILED };
  }

  redirect("/overview");
}

  const supabase = await createClient();
  await supabase.auth.signOut();
  redirect("/login");

// validation_errors is free-form JSON, so turn whatever is inside into readable lines.
export function formatValidationErrors(value: unknown): string[] {
  if (value === null || value === undefined) return [];

  const one = (item: unknown): string => {
    if (typeof item === "string") return item;
    if (item && typeof item === "object") {
      const o = item as { path?: unknown; message?: unknown };
      if (typeof o.message === "string") {
        const path = Array.isArray(o.path) ? o.path.join(".") : "";
        return path ? `${path}: ${o.message}` : o.message;
      }
    }
    return JSON.stringify(item);
  };

  if (Array.isArray(value)) return value.map(one);
  if (typeof value === "object") {
    return Object.entries(value as Record<string, unknown>).map(
      ([key, v]) => `${key}: ${typeof v === "string" ? v : JSON.stringify(v)}`,
    );
  }
  return [String(value)];
}

    .max(40),

  termsUrl: z.union([z.literal(""), z.url()]),
  notes: z.string().max(2000),

export async function createProvider(
  _previous: CreateProviderState,
  formData: FormData,
): Promise<CreateProviderState> {
  // First line of every Server Action.
  const admin = await requireAdmin();

  const values = {
    provider: String(formData.get("provider") ?? "").trim(),
    displayName: String(formData.get("displayName") ?? "").trim(),
    licenseStatus: String(formData.get("licenseStatus") ?? "unverified"),
    termsUrl: String(formData.get("termsUrl") ?? "").trim(),
    notes: String(formData.get("notes") ?? "").trim(),
  };

  const parsed = createSchema.safeParse(values);
  if (!parsed.success) {
    return {
      error:
        "Check the fields. The provider id may only use lowercase letters, numbers and dashes (2-40 characters).",
      message: null,
      values,
    };
  }

  const supabase = createSecretClient();
  const { error } = await supabase.from("providers").insert({
    provider: parsed.data.provider,
    display_name: parsed.data.displayName,
    license_status: parsed.data.licenseStatus,
    terms_url: parsed.data.termsUrl || null,
    notes: parsed.data.notes || null,
  });

  if (error) {
    if (error.code === "23505") {
      return { error: "A provider with this id already exists.", message: null, values };
    }
    console.error("createProvider failed:", error.message);
    return { error: "Could not add the provider.", message: null, values };
  }

  const { error: auditError } = await supabase.from("audit_logs").insert({
    admin_id: admin.id,
    action: "provider.create",
    entity_type: "provider",
    details: { provider: parsed.data.provider, license_status: parsed.data.licenseStatus },
  });

  revalidatePath("/providers");

  if (auditError) {
    console.error("audit log failed:", auditError.message);
    return {
      error: "The provider was added, but the audit log entry failed.",
      message: null,
      values: {},
    };
  }

  return { error: null, message: `Provider "${parsed.data.provider}" added.`, values: {} };
}

import { createServerClient } from "@supabase/ssr";
import { NextResponse, type NextRequest } from "next/server";
import { getPublicEnv } from "@/lib/env";

export async function proxy(request: NextRequest) {
  let response = NextResponse.next({ request });
  const { url, publishableKey } = getPublicEnv();

  const supabase = createServerClient(url, publishableKey, {
    cookies: {
      getAll() {
        return request.cookies.getAll();
      },
      setAll(cookiesToSet) {
        for (const { name, value } of cookiesToSet) request.cookies.set(name, value);
        response = NextResponse.next({ request });
        for (const { name, value, options } of cookiesToSet)
          response.cookies.set(name, value, options);
      },
    },
  });

  const {
    data: { user },
  } = await supabase.auth.getUser();

  if (!user && request.nextUrl.pathname !== "/login") {
    const loginUrl = request.nextUrl.clone();
    loginUrl.pathname = "/login";
    loginUrl.search = "";
    const redirectResponse = NextResponse.redirect(loginUrl);
    for (const cookie of response.cookies.getAll()) redirectResponse.cookies.set(cookie);
    return redirectResponse;
  }

  return response;
}

import type { ImportStats, StagingStore } from "@berryblue/provider-core";
import type { SupabaseClient } from "@supabase/supabase-js";

const PAGE_SIZE = 1000; // Supabase ek baar mein itni hi rows deta hai
const INSERT_CHUNK = 200;

// core ka StagingStore interface, asli Supabase database ke saath.
export function createSupabaseStagingStore(supabase: SupabaseClient): StagingStore {
  return {
    async startBatch(provider) {
      const { data, error } = await supabase
        .from("import_batches")
        .insert({ provider })
        .select("id")
        .single();

      if (error) throw new Error(`startBatch failed: ${error.message}`);
      return data.id as string;
    },

    async getKnownHashes(provider) {
      const known = new Map<string, Set<string>>();

      for (let from = 0; ; from += PAGE_SIZE) {
        const { data, error } = await supabase
          .from("staged_games")
          .select("provider_game_id, content_hash")
          .eq("provider", provider)
          .order("id")
          .range(from, from + PAGE_SIZE - 1);

        if (error) throw new Error(`getKnownHashes failed: ${error.message}`);

        for (const row of data) {
          const hashes = known.get(row.provider_game_id) ?? new Set<string>();
          hashes.add(row.content_hash);
          known.set(row.provider_game_id, hashes);
        }

        if (data.length < PAGE_SIZE) break;
      }

      return known;
    },

    async getCategoryMappings(provider) {
      const { data, error } = await supabase
        .from("category_mappings")
        .select("provider_category, category_id")
        .eq("provider", provider);

      if (error) throw new Error(`getCategoryMappings failed: ${error.message}`);

      return new Map<string, string>(
        data.map((row): [string, string] => [row.provider_category, row.category_id]),
      );
    },

    async insertStaged(batchId, rows) {
      for (let i = 0; i < rows.length; i += INSERT_CHUNK) {
        const chunk = rows.slice(i, i + INSERT_CHUNK).map((row) => ({
          batch_id: batchId,
          provider: row.provider,
          provider_game_id: row.providerGameId,
          raw: row.raw,
          slug: row.slug ?? null,
          title: row.title ?? null,
          description: row.description ?? null,
          thumbnail_url: row.thumbnailUrl ?? null,
          embed_url: row.embedUrl ?? null,
          provider_category: row.providerCategory ?? null,
          category_id: row.categoryId ?? null,
          tags: row.tags,
          devices: row.devices,
          content_hash: row.contentHash,
          status: row.status,
          validation_errors: row.validationErrors,
        }));

        // Wohi (provider, id, hash) pehle se ho to khamoshi se chhod do.
        const { error } = await supabase.from("staged_games").upsert(chunk, {
          onConflict: "provider,provider_game_id,content_hash",
          ignoreDuplicates: true,
        });

        if (error) throw new Error(`insertStaged failed: ${error.message}`);
      }
    },

    async finishBatch(batchId, stats: ImportStats) {
      const { error } = await supabase
        .from("import_batches")
        .update({
          status: "completed",
          finished_at: new Date().toISOString(),
          total_count: stats.total,
          new_count: stats.newCount,
          changed_count: stats.changedCount,
          unchanged_count: stats.unchangedCount,
          invalid_count: stats.invalidCount,
        })
        .eq("id", batchId);

      if (error) throw new Error(`finishBatch failed: ${error.message}`);
    },

    async failBatch(batchId, message) {
      const { error } = await supabase
        .from("import_batches")
        .update({ status: "failed", finished_at: new Date().toISOString(), error: message })
        .eq("id", batchId);

      // Yahan throw nahi: asli error pehle hi upar jaa raha hai, usay chhupana nahi.
      if (error) console.error(`failBatch failed: ${error.message}`);
    },
  };
}

===== FILE: packages\providers\core\src\hash.ts =====
// Keys ki tarteeb se qata nazar ek hi data ka ek hi text banata hai.
function stableStringify(value: unknown): string {
  if (value === null || typeof value !== "object") return JSON.stringify(value) ?? "null";

  if (Array.isArray(value)) return "[" + value.map(stableStringify).join(",") + "]";

  const entries = Object.entries(value as Record<string, unknown>)
    .filter(([, v]) => v !== undefined)
    .sort(([a], [b]) => (a < b ? -1 : a > b ? 1 : 0))
    .map(([k, v]) => JSON.stringify(k) + ":" + stableStringify(v));

  return "{" + entries.join(",") + "}";
}

// SHA-256 fingerprint: provider ka data badle to hash badalta hai, warna wahi rehta hai.
export async function contentHash(value: unknown): Promise<string> {
  const bytes = new TextEncoder().encode(stableStringify(value));
  const digest = await crypto.subtle.digest("SHA-256", bytes);
  return Array.from(new Uint8Array(digest), (b) => b.toString(16).padStart(2, "0")).join("");
}

import { contentHash } from "./hash";
import { normalizeProviderCategory } from "./normalize";
import type { ImportStats, ProviderAdapter, StagedRow, StagingStore } from "./types";
import { validateDraft } from "./validate";

export type ImportResult = {
  batchId: string;
  stats: ImportStats;
  // Woh games jo staging tak pohonch hi nahi sake (adapter ne throw kiya ya id nahi thi).
  skipped: { index: number; reason: string }[];
};

export async function runImport(
  adapter: ProviderAdapter,
  store: StagingStore,
): Promise<ImportResult> {
  const batchId = await store.startBatch(adapter.provider);

  const stats: ImportStats = {
    total: 0,
    newCount: 0,
    changedCount: 0,
    unchangedCount: 0,
    invalidCount: 0,
  };
  const skipped: ImportResult["skipped"] = [];

  try {
    const catalog = await adapter.fetchCatalog();
    const known = await store.getKnownHashes(adapter.provider);
    const categoryMap = await store.getCategoryMappings(adapter.provider);
    const seenInFeed = new Set<string>();
    const rows: StagedRow[] = [];

    for (const [index, raw] of catalog.entries()) {
      stats.total += 1;

      let draft;
      try {
        draft = adapter.toInternal(raw);
      } catch (error) {
        stats.invalidCount += 1;
        skipped.push({ index, reason: error instanceof Error ? error.message : String(error) });
        continue;
      }

      if (!draft.providerGameId) {
        stats.invalidCount += 1;
        skipped.push({ index, reason: "providerGameId missing" });
        continue;
      }

      const hash = await contentHash(raw);
      const key = draft.providerGameId + ":" + hash;

      // Wohi game feed mein do baar aaye, ya pehle hi staging mein hai: dobara nahi.
      const hashes = known.get(draft.providerGameId);
      if (seenInFeed.has(key) || hashes?.has(hash)) {
        stats.unchangedCount += 1;
        continue;
      }
      seenInFeed.add(key);

      if (hashes === undefined) stats.newCount += 1;
      else stats.changedCount += 1;

      const base = {
        provider: adapter.provider,
        providerGameId: draft.providerGameId,
        raw,
        providerCategory: draft.providerCategory,
        categoryId:
          categoryMap.get(normalizeProviderCategory(draft.providerCategory) ?? "") ?? null,
        contentHash: hash,
      };

      const validation = validateDraft(draft);

      if (validation.ok) {
        rows.push({ ...base, ...validation.data, status: "pending", validationErrors: null });
      } else {
        stats.invalidCount += 1;
        rows.push({
          ...base,
          slug: draft.slug,
          title: draft.title,
          description: draft.description,
          thumbnailUrl: draft.thumbnailUrl,
          embedUrl: draft.embedUrl,
          tags: draft.tags ?? [],
          devices: draft.devices ?? [],
          status: "invalid",
          validationErrors: validation.errors,
        });
      }
    }

    if (rows.length > 0) await store.insertStaged(batchId, rows);
    await store.finishBatch(batchId, stats);

    return { batchId, stats, skipped };
  } catch (error) {
    await store.failBatch(batchId, error instanceof Error ? error.message : String(error));
    throw error;
  }
}

-- Approve = game live (insert ya update), staged row "approved", audit log: sab ek transaction mein.
create function public.approve_staged_game(
  p_staged_id uuid,
  p_category_id uuid,
  p_admin_id uuid,
  p_slug text default null
)
returns uuid
language plpgsql
set search_path = ''
as $$
declare
  s public.staged_games%rowtype;
  v_slug text;
  v_game_id uuid;
begin
  select * into s from public.staged_games where id = p_staged_id for update;

  if not found then
    raise exception 'staged game not found';
  end if;
  if s.status <> 'pending' then
    raise exception 'only pending games can be approved (current status: %)', s.status;
  end if;
  if not exists (select 1 from public.categories where id = p_category_id) then
    raise exception 'category not found';
  end if;

  v_slug := coalesce(nullif(btrim(p_slug), ''), s.slug);
  if v_slug is null or v_slug !~ '^[a-z0-9-]+$' then
    raise exception 'invalid slug';
  end if;

  -- Takraav: yeh slug kisi AUR game ka hai (wohi provider + wohi id ho to woh update hai, takraav nahi).
  if exists (
    select 1 from public.games g
    where g.slug = v_slug
      and not (g.provider = s.provider and g.provider_game_id = s.provider_game_id)
  ) then
    raise exception 'slug "%" is already used by another game', v_slug;
  end if;

  insert into public.games (
    slug, title, description, thumbnail_url, embed_url, category_id,
    tags, devices, provider, provider_game_id, content_hash, is_published
  )
  values (
    v_slug, s.title, s.description, s.thumbnail_url, s.embed_url, p_category_id,
    s.tags, s.devices, s.provider, s.provider_game_id, s.content_hash, true
  )
  on conflict (provider, provider_game_id) do update set
    slug = excluded.slug,
    title = excluded.title,
    description = excluded.description,
    thumbnail_url = excluded.thumbnail_url,
    embed_url = excluded.embed_url,
    category_id = excluded.category_id,
    tags = excluded.tags,
    devices = excluded.devices,
    content_hash = excluded.content_hash,
    is_published = true
  returning id into v_game_id;

  update public.staged_games
  set status = 'approved',
      slug = v_slug,
      category_id = p_category_id,
      reviewed_by = p_admin_id,
      reviewed_at = now()
  where id = p_staged_id;

  insert into public.audit_logs (admin_id, action, entity_type, entity_id, details)
  values (
    p_admin_id, 'game.approve', 'game', v_game_id,
    jsonb_build_object('staged_id', p_staged_id, 'slug', v_slug, 'provider', s.provider)
  );

  return v_game_id;
end;
$$;
