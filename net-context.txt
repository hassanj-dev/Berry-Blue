"use client";

import { SignalHigh, SignalLow } from "lucide-react";
import { t } from "@/i18n";
import { cn } from "@/lib/cn";
import { setLowInternet, useLowInternet } from "@/lib/low-internet";

export function LowInternetToggle() {
  const on = useLowInternet();
  const Signal = on ? SignalLow : SignalHigh;

  return (
    <button
      type="button"
      aria-pressed={on}
      aria-label={t("lowInternet.label")}
      title={t("lowInternet.title")}
      onClick={() => setLowInternet(!on)}
      className={cn(
        "flex size-10 shrink-0 cursor-pointer items-center justify-center rounded-full transition-colors hover:text-foreground",
        on ? "text-brand" : "text-muted",
      )}
    >
      <Signal size={20} strokeWidth={1.5} aria-hidden="true" />
    </button>
  );
}

===== apps\web\lib\low-internet-script.ts =====
export const LOW_INTERNET_KEY = "low-internet";
export const LOW_INTERNET_ATTR = "data-low-internet";

// Runs in <head> before the page is painted. Order of decisions:
// 1. the visitor's own choice ("on" or "off") wins,
// 2. otherwise Save-Data or a 2g connection turns the mode on.
export const lowInternetInitScript = `(function(){var on=false;try{var s=localStorage.getItem("${LOW_INTERNET_KEY}");if(s==="on"){on=true}else if(s!=="off"){var c=navigator.connection;on=!!c&&(c.saveData===true||c.effectiveType==="2g"||c.effectiveType==="slow-2g")}}catch(e){}if(on){document.documentElement.setAttribute("${LOW_INTERNET_ATTR}","")}})();`;

===== apps\web\lib\low-internet.ts =====
import { useSyncExternalStore } from "react";
import { LOW_INTERNET_ATTR, LOW_INTERNET_KEY } from "@/lib/low-internet-script";

const listeners = new Set<() => void>();

function emit() {
  for (const listener of listeners) listener();
}

// Another tab changed the choice: follow it.
function onStorage(event: StorageEvent) {
  if (event.key !== LOW_INTERNET_KEY) return;
  document.documentElement.toggleAttribute(LOW_INTERNET_ATTR, event.newValue === "on");
  emit();
}

function subscribe(listener: () => void) {
  if (listeners.size === 0) window.addEventListener("storage", onStorage);
  listeners.add(listener);

  return () => {
    listeners.delete(listener);
    if (listeners.size === 0) window.removeEventListener("storage", onStorage);
  };
}

function getSnapshot() {
  return document.documentElement.hasAttribute(LOW_INTERNET_ATTR);
}

// false on the server and while hydrating, so the button never causes a hydration warning.
function getServerSnapshot() {
  return false;
}

export function useLowInternet() {
  return useSyncExternalStore(subscribe, getSnapshot, getServerSnapshot);
}

export function setLowInternet(on: boolean) {
  document.documentElement.toggleAttribute(LOW_INTERNET_ATTR, on);
  try {
    localStorage.setItem(LOW_INTERNET_KEY, on ? "on" : "off");
  } catch {
    // Storage can be blocked (private mode). The mode still works until the page closes.
  }
  emit();
}
