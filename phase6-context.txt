type Params = Promise<{ slug: string }>;

export async function generateMetadata({ params }: { params: Params }): Promise<Metadata> {
  const { slug } = await params;
  const game = await getGameBySlug(slug);

  if (!game) return { title: t("game.notFound") };

  return {
    title: game.title,
    description: game.description || undefined,
  };
}

// The page itself asks for no data, it only adds Suspense. The data is read inside GameContent.
export default function GamePage({ params }: { params: Params }) {
  return (
    <main>
      <Suspense fallback={<GamePageSkeleton />}>
        <GameContent params={params} />
      </Suspense>
    </main>
  );
}

"use client";

import { Maximize, Minimize, Play } from "lucide-react";
import Image from "next/image";
import { useRef, useState, useSyncExternalStore, type ReactNode } from "react";
import { actionButton } from "@/components/game/action-styles";
import { ThumbBlock } from "@/components/game/ThumbBlock";
import { t } from "@/i18n";
import { recordPlay } from "@/lib/play-actions";
import { addRecent } from "@/lib/recent";
import { cn } from "@/lib/cn";

type GamePlayerProps = {
  title: string;
  thumbnailUrl: string;
  embedUrl: string;
  // First row under the player: play count, like and dislike.
  stats?: ReactNode;
  // Second row, left side: favorite and share.
  actions?: ReactNode;
  // Second row, right side, before the fullscreen button: report and phone QR.
  meta?: ReactNode;
  gameId: string;
  slug: string;
};

// sandbox = the least permission the game needs inside the iframe.
// allow-same-origin is only given to games on another domain: for our own games,
// allow-scripts together with allow-same-origin would defeat the sandbox.
function sandboxFor(embedUrl: string) {
  const base = "allow-scripts allow-pointer-lock";
  return embedUrl.startsWith("/") ? base : `${base} allow-same-origin`;
}

// The browser owns the fullscreen state, so we subscribe to it instead of copying it into
// our own state. This keeps the button correct even when the user presses Esc.
function subscribeFullscreen(onChange: () => void) {
  document.addEventListener("fullscreenchange", onChange);
  return () => document.removeEventListener("fullscreenchange", onChange);
}

function subscribeNothing() {
  return () => {};
}

export function GamePlayer({
  gameId,
  slug,
  title,
  thumbnailUrl,
  embedUrl,
  stats,
  actions,
  meta,
}: GamePlayerProps) {
  const [playing, setPlaying] = useState(false);
  const boxRef = useRef<HTMLDivElement>(null);

  const isFullscreen = useSyncExternalStore(
    subscribeFullscreen,
    () => document.fullscreenElement !== null,
    () => false,
  );
  // false on the server, so the button appears after hydration. iPhone Safari stays false.
  const canFullscreen = useSyncExternalStore(
    subscribeNothing,
    () => document.fullscreenEnabled,
    () => false,
  );

  function handlePlay() {
    setPlaying(true);
    addRecent(slug);

    // Count each game once per browser session (a page refresh does not count again).
    const key = "played:" + gameId;
    try {
      if (sessionStorage.getItem(key)) return;
      sessionStorage.setItem(key, "1");
    } catch {
      // Storage can be blocked (private mode). The play is still counted.
    }
    void recordPlay(gameId);
  }

  async function toggleFullscreen() {
    try {
      if (document.fullscreenElement) {
        await document.exitFullscreen();
      } else {
        await boxRef.current?.requestFullscreen();
      }
    } catch (error) {
      console.error("Fullscreen failed:", error);
    }
  }

  const fullscreenLabel = isFullscreen ? t("game.exitFullscreen") : t("game.fullscreen");

  return (
    <div>
      <div
        ref={boxRef}
        className={cn(
          "relative aspect-video w-full overflow-hidden border border-border bg-surface",
          isFullscreen ? "rounded-none border-0" : "rounded-2xl",
        )}
      >
        {playing ? (
          <iframe
            src={embedUrl}
            title={t("game.playerTitle", { title })}
            sandbox={sandboxFor(embedUrl)}
            allow="fullscreen"
            allowFullScreen
            className="absolute inset-0 size-full border-0"
          />
        ) : (
          <>
            <Image
              src={thumbnailUrl}
              alt=""
              fill
              priority
              sizes="(min-width: 1152px) 1152px, 100vw"
              className="card-thumb object-cover"
            />
            <ThumbBlock title={title} className="absolute inset-0 text-7xl" />
            <div className="absolute inset-0 flex items-center justify-center bg-background/50">
              <button
                type="button"
                onClick={handlePlay}
                className="inline-flex cursor-pointer items-center gap-2 rounded-full bg-primary px-8 py-4 text-lg font-medium text-primary-foreground max-sm:px-6 max-sm:py-3 max-sm:text-base transition-colors hover:bg-primary-hover"
              >
                <Play size={20} strokeWidth={1.5} fill="currentColor" aria-hidden="true" />
                {t("game.playNow")}
              </button>
            </div>
          </>
        )}

        {/* Inside the player box, so it is also there in fullscreen, before and during play. */}
        {isFullscreen && (
          <button
            type="button"
            onClick={toggleFullscreen}
            aria-label={t("game.exitFullscreen")}
            title={t("game.exitFullscreen")}
            className="absolute right-3 top-3 z-10 flex size-11 cursor-pointer items-center justify-center rounded-full bg-black/60 text-white transition-colors hover:bg-black/80"
          >
            <Minimize size={20} strokeWidth={1.5} aria-hidden="true" />
          </button>
        )}
      </div>

      <div className="relative mt-3 flex flex-wrap items-center justify-between gap-2 sm:mt-4 sm:justify-start">
        <div className="flex flex-wrap items-center gap-2">{stats}</div>
        <div className="flex items-center gap-2 max-sm:ml-auto max-sm:gap-1.5 sm:flex-1 sm:justify-start">
          {actions}
          <div className="contents sm:ml-auto sm:flex sm:items-center sm:gap-2">
            {meta}
            {canFullscreen && (
              <button
                type="button"
                onClick={toggleFullscreen}
                aria-label={fullscreenLabel}
                title={fullscreenLabel}
                className={actionButton}
              >
                {isFullscreen ? (
                  <Minimize size={18} strokeWidth={1.5} aria-hidden="true" />
                ) : (
                  <Maximize size={18} strokeWidth={1.5} aria-hidden="true" />
                )}
              </button>
            )}
          </div>
        </div>
      </div>
    </div>
  );
}

"use client";

import { Share2, Smartphone, X } from "lucide-react";
import { useEffect, useRef, useState, useSyncExternalStore } from "react";
import { actionButton } from "@/components/game/action-styles";
import { t } from "@/i18n";

type Status = "idle" | "copied" | "failed";

// "Desktop" = a mouse-like device with a wide screen. Phones and tablets do not get the QR code.
const DESKTOP_QUERY = "(hover: hover) and (pointer: fine) and (min-width: 1024px)";

function subscribeDesktop(onChange: () => void) {
  const query = window.matchMedia(DESKTOP_QUERY);
  query.addEventListener("change", onChange);
  return () => query.removeEventListener("change", onChange);
}

function getDesktop() {
  return window.matchMedia(DESKTOP_QUERY).matches;
}

// No query string or hash, so shared links stay clean.
function pageUrl() {
  return window.location.origin + window.location.pathname;
}

// Round share button for the toolbar. The "Link copied" note appears next to it.
export function ShareButton({ title }: { title: string }) {
  const [status, setStatus] = useState<Status>("idle");
  const timer = useRef<number | undefined>(undefined);

  useEffect(() => () => window.clearTimeout(timer.current), []);

  function showStatus(next: Status) {
    setStatus(next);
    window.clearTimeout(timer.current);
    timer.current = window.setTimeout(() => setStatus("idle"), 3000);
  }

  async function handleShare() {
    const url = pageUrl();

    if (typeof navigator.share === "function") {
      try {
        await navigator.share({ title, text: t("game.shareText", { title }), url });
        return;
      } catch (error) {
        // The visitor closed the share sheet: nothing to do. Any other error falls back to copying.
        if (error instanceof DOMException && error.name === "AbortError") return;
      }
    }

    try {
      await navigator.clipboard.writeText(url);
      showStatus("copied");
    } catch {
      showStatus("failed");
    }
  }

  const statusText =
    status === "copied" ? t("game.linkCopied") : status === "failed" ? t("game.copyFailed") : "";

  return (
    <>
      <button
        type="button"
        onClick={handleShare}
        aria-label={t("game.share")}
        title={t("game.share")}
        className={actionButton}
      >
        <Share2 size={18} strokeWidth={1.5} aria-hidden="true" />
      </button>
      <span
        aria-live="polite"
        className="text-sm text-muted empty:hidden max-sm:absolute max-sm:right-0 max-sm:top-full max-sm:z-20 max-sm:mt-2 max-sm:rounded-lg max-sm:border max-sm:border-border max-sm:bg-surface max-sm:px-3 max-sm:py-1.5"
      >
        {statusText}
      </span>
    </>
  );
}

// Phone icon in the toolbar (desktop only). It opens a small popup with the QR code.
// The QR library loads only when the popup is opened.
export function GameQr() {
  const isDesktop = useSyncExternalStore(subscribeDesktop, getDesktop, () => false);
  const [qr, setQr] = useState<string | null>(null);
  const dialogRef = useRef<HTMLDialogElement>(null);

  async function openPopup() {
    dialogRef.current?.showModal();
    if (qr) return;

    try {
      const QRCode = await import("qrcode");
      setQr(
        await QRCode.toDataURL(pageUrl(), {
          width: 288,
          margin: 2,
          errorCorrectionLevel: "M",
        }),
      );
    } catch (error) {
      console.error("QR code failed:", error);
    }
  }

  if (!isDesktop) return null;

  return (
    <>
      <button
        type="button"
        onClick={openPopup}
        aria-label={t("game.playOnPhone")}
        title={t("game.playOnPhone")}
        className={actionButton}
      >
        <Smartphone size={18} strokeWidth={1.5} aria-hidden="true" />
      </button>

      <dialog
        ref={dialogRef}
        aria-label={t("game.playOnPhone")}
        // A click on the dark area around the popup lands on the dialog itself, and closes it.
        onClick={(event) => {
          if (event.target === event.currentTarget) event.currentTarget.close();
        }}
        className="m-auto w-72 max-w-[calc(100vw-2rem)] rounded-2xl border border-border bg-surface p-0 text-foreground backdrop:bg-black/60"
      >
        <button
          type="button"
          onClick={() => dialogRef.current?.close()}
          aria-label={t("common.close")}
          title={t("common.close")}
          className="absolute right-2 top-2 flex size-9 cursor-pointer items-center justify-center rounded-full text-muted transition-colors hover:text-foreground"
        >
          <X size={18} strokeWidth={1.5} aria-hidden="true" />
        </button>

        <div className="flex flex-col items-center gap-4 p-6 pt-10 text-center">
          <div className="size-36 overflow-hidden rounded-xl bg-white">
            {qr ? (
              // A data URL cannot go through next/image, so a plain img is correct here.
              // eslint-disable-next-line @next/next/no-img-element
              <img src={qr} width={144} height={144} alt={t("game.qrAlt")} />
            ) : null}
          </div>
          <div className="flex flex-col gap-1">
            <p className="font-medium">{t("game.playOnPhone")}</p>
            <p className="text-sm text-muted">{t("game.scanQr")}</p>
          </div>
        </div>
      </dialog>
    </>
  );
}

import { Eye, Play } from "lucide-react";
import { t } from "@/i18n";
import { cn } from "@/lib/cn";
import { getPlayCount } from "@/lib/play-count";

const compact = new Intl.NumberFormat("en-US", { notation: "compact", maximumFractionDigits: 1 });

type PlayCountProps = {
  gameId: string;
  // "eye" is used in the phone info row under the title.
  icon?: "play" | "eye";
  className?: string;
};

export async function PlayCount({ gameId, icon = "play", className }: PlayCountProps) {
  const count = await getPlayCount(gameId);
  if (count === null) return null;

  const label = t("game.plays", { count: count.toLocaleString("en-US") });
  const Glyph = icon === "eye" ? Eye : Play;

  return (
    <p
      className={cn("flex items-center gap-1.5 text-sm tabular-nums text-muted", className)}
      title={label}
    >
      <Glyph
        size={16}
        strokeWidth={1.5}
        fill={icon === "play" ? "currentColor" : "none"}
        aria-hidden="true"
      />
      <span aria-hidden="true">{compact.format(count)}</span>
      <span className="sr-only">{label}</span>
    </p>
  );
}
