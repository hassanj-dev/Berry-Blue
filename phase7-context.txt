export async function generateMetadata({ params }: { params: Params }): Promise<Metadata> {
  const { slug } = await params;
  const game = await getGameBySlug(slug);

  if (!game) return { title: t("game.notFound") };

  return {
    title: game.title,
    description: game.description || undefined,
  };
}

import { Maximize, Minimize, Play } from "lucide-react";
import Image from "next/image";
import { useRef, useState, useSyncExternalStore, type ReactNode } from "react";
import { actionButton } from "@/components/game/action-styles";
import { ThumbBlock } from "@/components/game/ThumbBlock";
import { t } from "@/i18n";
import { recordPlay } from "@/lib/play-actions";
import { addRecent } from "@/lib/recent";
import { cn } from "@/lib/cn";

type GamePlayerProps = {
  title: string;
  thumbnailUrl: string;
  embedUrl: string;
  // First row under the player: play count, like and dislike.
  stats?: ReactNode;
  // Second row, left side: favorite and share.
  actions?: ReactNode;
  // Second row, right side, before the fullscreen button: report and phone QR.
  meta?: ReactNode;
  gameId: string;
  slug: string;
};

// sandbox = the least permission the game needs inside the iframe.
// allow-same-origin is only given to games on another domain: for our own games,
// allow-scripts together with allow-same-origin would defeat the sandbox.
function sandboxFor(embedUrl: string) {
  const base = "allow-scripts allow-pointer-lock";
  return embedUrl.startsWith("/") ? base : `${base} allow-same-origin`;
}

// The browser owns the fullscreen state, so we subscribe to it instead of copying it into
// our own state. This keeps the button correct even when the user presses Esc.
function subscribeFullscreen(onChange: () => void) {
  document.addEventListener("fullscreenchange", onChange);
  return () => document.removeEventListener("fullscreenchange", onChange);
}

function subscribeNothing() {
  return () => {};
}

export function GamePlayer({
  gameId,
  slug,
  title,
  thumbnailUrl,
  embedUrl,
  stats,
  actions,
  meta,
}: GamePlayerProps) {
  const [playing, setPlaying] = useState(false);
  const boxRef = useRef<HTMLDivElement>(null);

  const isFullscreen = useSyncExternalStore(
    subscribeFullscreen,
    () => document.fullscreenElement !== null,
    () => false,
  );
  // false on the server, so the button appears after hydration. iPhone Safari stays false.
  const canFullscreen = useSyncExternalStore(
    subscribeNothing,
    () => document.fullscreenEnabled,
    () => false,
  );

  function handlePlay() {
    setPlaying(true);
    addRecent(slug);

    // Count each game once per browser session (a page refresh does not count again).
    const key = "played:" + gameId;
    try {
      if (sessionStorage.getItem(key)) return;
      sessionStorage.setItem(key, "1");
    } catch {
      // Storage can be blocked (private mode). The play is still counted.
    }
    void recordPlay(gameId);
  }

  async function toggleFullscreen() {
    try {
      if (document.fullscreenElement) {
        await document.exitFullscreen();
      } else {
        await boxRef.current?.requestFullscreen();
      }
    } catch (error) {
      console.error("Fullscreen failed:", error);
    }
  }

  const fullscreenLabel = isFullscreen ? t("game.exitFullscreen") : t("game.fullscreen");

  return (
    <div>
      <div
        ref={boxRef}
        className={cn(
          "relative aspect-video w-full overflow-hidden border border-border bg-surface",
          isFullscreen ? "rounded-none border-0" : "rounded-2xl",
        )}
      >
        {playing ? (
          <iframe
            src={embedUrl}
            title={t("game.playerTitle", { title })}
            sandbox={sandboxFor(embedUrl)}
            allow="fullscreen"
            allowFullScreen
            className="absolute inset-0 size-full border-0"
          />
        ) : (
          <>
            <Image
              src={thumbnailUrl}
              alt=""
              fill
              priority
              sizes="(min-width: 1152px) 1152px, 100vw"
              className="card-thumb object-cover"
            />
            <ThumbBlock title={title} className="absolute inset-0 text-7xl" />
            <div className="absolute inset-0 flex items-center justify-center bg-background/50">
              <button
                type="button"
                onClick={handlePlay}
                className="inline-flex cursor-pointer items-center gap-2 rounded-full bg-primary px-8 py-4 text-lg font-medium text-primary-foreground shadow-lg shadow-primary/30 ring-1 ring-white/10 transition-[transform,background-color,box-shadow] duration-150 hover:bg-primary-hover hover:shadow-primary/40 active:scale-[0.97] max-sm:px-5 max-sm:py-2.5 max-sm:text-[15px] motion-reduce:transition-none motion-reduce:active:scale-100"
              >
                <Play size={20} strokeWidth={1.5} fill="currentColor" aria-hidden="true" />
                {t("game.playNow")}
              </button>
            </div>
          </>
        )}

        {/* Desktop: the enter-fullscreen button sits on the player, bottom right. Phones keep it in the toolbar. */}
        {canFullscreen && !isFullscreen && (
          <button
            type="button"
            onClick={toggleFullscreen}
            aria-label={fullscreenLabel}
            title={fullscreenLabel}
            className="absolute bottom-3 right-3 z-10 flex size-10 cursor-pointer items-center justify-center rounded-full bg-black/60 text-white transition-colors hover:bg-black/80 max-sm:hidden"
          >
            <Maximize size={18} strokeWidth={1.5} aria-hidden="true" />
          </button>
        )}

        {/* Inside the player box, so it is also there in fullscreen, before and during play. */}
        {isFullscreen && (
          <button
            type="button"
            onClick={toggleFullscreen}
            aria-label={t("game.exitFullscreen")}
            title={t("game.exitFullscreen")}
            className="absolute right-3 top-3 z-10 flex size-11 cursor-pointer items-center justify-center rounded-full bg-black/60 text-white transition-colors hover:bg-black/80"
          >
            <Minimize size={20} strokeWidth={1.5} aria-hidden="true" />
          </button>
        )}
      </div>

      <div className="relative mt-3 flex flex-wrap items-center justify-between gap-2 sm:mt-4 sm:justify-start">
        <div className="flex flex-wrap items-center gap-2">{stats}</div>
        <div className="flex items-center gap-2 max-sm:ml-auto max-sm:gap-1.5 sm:flex-1 sm:justify-start">
          {actions}
          <div className="contents sm:ml-auto sm:flex sm:items-center sm:gap-2">
            {meta}
            {canFullscreen && (
              <button
                type="button"
                onClick={toggleFullscreen}
                aria-label={fullscreenLabel}
                title={fullscreenLabel}
                className={actionButton + " sm:hidden"}
              >

  );

"use client";

import { Share2, Smartphone, X } from "lucide-react";
import { useEffect, useRef, useState, useSyncExternalStore } from "react";
import { actionButton } from "@/components/game/action-styles";
import { t } from "@/i18n";

type Status = "idle" | "copied" | "failed";

// "Desktop" = a mouse-like device with a wide screen. Phones and tablets do not get the QR code.
const DESKTOP_QUERY = "(hover: hover) and (pointer: fine) and (min-width: 1024px)";

function subscribeDesktop(onChange: () => void) {
  const query = window.matchMedia(DESKTOP_QUERY);
  query.addEventListener("change", onChange);
  return () => query.removeEventListener("change", onChange);
}

function getDesktop() {
  return window.matchMedia(DESKTOP_QUERY).matches;
}

// No query string or hash, so shared links stay clean.
function pageUrl() {
  return window.location.origin + window.location.pathname;
}

// Round share button for the toolbar. The "Link copied" note appears next to it.
export function ShareButton({ title }: { title: string }) {
  const [status, setStatus] = useState<Status>("idle");
  const timer = useRef<number | undefined>(undefined);

  useEffect(() => () => window.clearTimeout(timer.current), []);

  function showStatus(next: Status) {
    setStatus(next);
    window.clearTimeout(timer.current);
    timer.current = window.setTimeout(() => setStatus("idle"), 3000);
  }

  async function handleShare() {
    const url = pageUrl();

    if (typeof navigator.share === "function") {
      try {
        await navigator.share({ title, text: t("game.shareText", { title }), url });
        return;
      } catch (error) {
        // The visitor closed the share sheet: nothing to do. Any other error falls back to copying.
        if (error instanceof DOMException && error.name === "AbortError") return;
      }
    }

    try {
      await navigator.clipboard.writeText(url);
      showStatus("copied");
    } catch {
      showStatus("failed");
    }
  }

  const statusText =
    status === "copied" ? t("game.linkCopied") : status === "failed" ? t("game.copyFailed") : "";

  return (
    <>
      <button
        type="button"
        onClick={handleShare}
        aria-label={t("game.share")}
        title={t("game.share")}
        className={actionButton}
      >
        <Share2 size={18} strokeWidth={1.5} aria-hidden="true" />
      </button>
      <span
        aria-live="polite"
        className="absolute right-0 top-full z-20 mt-2 rounded-lg border border-border bg-surface px-3 py-1.5 text-sm text-muted empty:hidden"
      >
        {statusText}
      </span>
    </>
  );
}

// Phone icon in the toolbar (desktop only). It opens a small popup with the QR code.
// The QR library loads only when the popup is opened.
export function GameQr() {
  const isDesktop = useSyncExternalStore(subscribeDesktop, getDesktop, () => false);
  const [qr, setQr] = useState<string | null>(null);
  const dialogRef = useRef<HTMLDialogElement>(null);

  async function openPopup() {
    dialogRef.current?.showModal();
    if (qr) return;

    try {
      const QRCode = await import("qrcode");
      setQr(
        await QRCode.toDataURL(pageUrl(), {
          width: 288,
          margin: 2,
          errorCorrectionLevel: "M",
        }),
      );
    } catch (error) {
      console.error("QR code failed:", error);
    }
  }

  if (!isDesktop) return null;

  return (
    <>
      <button
        type="button"
        onClick={openPopup}
        aria-label={t("game.playOnPhone")}
        title={t("game.playOnPhone")}
        className={actionButton}
      >
        <Smartphone size={18} strokeWidth={1.5} aria-hidden="true" />
      </button>

      <dialog
        ref={dialogRef}
        aria-label={t("game.playOnPhone")}
        // A click on the dark area around the popup lands on the dialog itself, and closes it.
        onClick={(event) => {
          if (event.target === event.currentTarget) event.currentTarget.close();
        }}
        className="m-auto w-72 max-w-[calc(100vw-2rem)] rounded-2xl border border-border bg-surface p-0 text-foreground backdrop:bg-black/60"
      >
        <button
          type="button"
          onClick={() => dialogRef.current?.close()}
          aria-label={t("common.close")}
          title={t("common.close")}
          className="absolute right-2 top-2 flex size-9 cursor-pointer items-center justify-center rounded-full text-muted transition-colors hover:text-foreground"
        >
          <X size={18} strokeWidth={1.5} aria-hidden="true" />
        </button>

        <div className="flex flex-col items-center gap-4 p-6 pt-10 text-center">
          <div className="size-36 overflow-hidden rounded-xl bg-white">
            {qr ? (
              // A data URL cannot go through next/image, so a plain img is correct here.
              // eslint-disable-next-line @next/next/no-img-element
              <img src={qr} width={144} height={144} alt={t("game.qrAlt")} />
            ) : null}
          </div>
          <div className="flex flex-col gap-1">
            <p className="font-medium">{t("game.playOnPhone")}</p>
            <p className="text-sm text-muted">{t("game.scanQr")}</p>
          </div>
        </div>
      </dialog>
    </>
  );
}

// A flag button. It opens a small panel under the toolbar (the toolbar is the positioned parent).
export function GameReport({ gameId }: { gameId: string }) {
  const [reason, setReason] = useState("not_loading");
  const [status, setStatus] = useState<"idle" | "sent" | "failed">("idle");
  const [pending, startTransition] = useTransition();

  function send() {
    startTransition(async () => {
      try {
        setStatus((await reportGame(gameId, reason)) ? "sent" : "failed");
      } catch {
        setStatus("failed");
      }
    });
  }

  if (status === "sent") {
    return <p className="text-sm text-muted">{t("game.reportSent")}</p>;
  }

  return (
    <details className="text-sm">
      <summary
        aria-label={t("game.report")}

      </summary>

        <select
          aria-label={t("game.reportReason")}
          value={reason}
          onChange={(event) => setReason(event.target.value)}
          className="h-11 rounded-lg border border-border bg-background px-3"

import { Star } from "lucide-react";
import { t } from "@/i18n";
import { cn } from "@/lib/cn";
import { getVoteCounts } from "@/lib/vote-count";

// Star rating out of 5, worked out from likes and dislikes. Hidden until the first vote.
// A small "starting weight" (PRIOR_VOTES votes at PRIOR_SCORE) stops one single vote from
// showing 5.0 or 1.0. With many votes the real like share takes over.
// Examples: 1 like = 4.7, 10 likes = 4.9, 8 likes + 2 dislikes = 4.1, 1 dislike = 3.0.
const PRIOR_VOTES = 2;
const PRIOR_SCORE = 0.9;

// It is read when the page loads, so after a vote it updates on the next load.
export async function GameRating({ gameId, className }: { gameId: string; className?: string }) {
  const counts = await getVoteCounts(gameId);
  if (!counts) return null;

  const total = counts.likes + counts.dislikes;
  if (total === 0) return null;

  const share = (counts.likes + PRIOR_VOTES * PRIOR_SCORE) / (total + PRIOR_VOTES);
  const rating = Math.min(5, Math.round(share * 50) / 10).toFixed(1);

  // The tooltip and screen readers still get the exact like percentage and vote count.
  const percent = Math.round((counts.likes / total) * 100);
  const summary = t("game.voteSummary", {
    percent: String(percent),
    count: total.toLocaleString("en-US"),
  });

  return (
    <p
      className={cn("flex items-center gap-1.5 text-sm tabular-nums text-muted", className)}
      title={summary}
    >
      <Star
        size={16}
        strokeWidth={1.5}
        fill="currentColor"
        className="text-amber-400"
        aria-hidden="true"
      />
      <span aria-hidden="true">{rating}</span>
      <span className="sr-only">{summary}</span>
    </p>
  );
}

"use client";

import { ThumbsDown, ThumbsUp } from "lucide-react";
import { useState, useSyncExternalStore, useTransition } from "react";
import { actionButton } from "@/components/game/action-styles";
import { t } from "@/i18n";
import type { VoteCounts } from "@/lib/vote-count";
import { castVote } from "@/lib/vote-actions";

type Vote = -1 | 0 | 1;

// The visitor's own vote lives in localStorage. The memory copy is the fallback when storage is blocked.
const memory = new Map<string, Vote>();
const listeners = new Set<() => void>();

function storageKey(gameId: string) {
  return "vote:" + gameId;
}

function readVote(gameId: string): Vote {
  try {
    const value = localStorage.getItem(storageKey(gameId));
    return value === "1" ? 1 : value === "-1" ? -1 : 0;
  } catch {
    return memory.get(gameId) ?? 0;
  }
}

function saveVote(gameId: string, vote: Vote) {
  memory.set(gameId, vote);
  try {
    if (vote === 0) localStorage.removeItem(storageKey(gameId));
    else localStorage.setItem(storageKey(gameId), String(vote));
  } catch {
    // Storage can be blocked (private mode). The memory copy keeps the vote for this visit.
  }
  listeners.forEach((listener) => listener());
}

function subscribe(listener: () => void) {
  listeners.add(listener);
  window.addEventListener("storage", listener);
  return () => {
    listeners.delete(listener);
    window.removeEventListener("storage", listener);
  };
}

function applyVote(counts: VoteCounts, from: Vote, to: Vote): VoteCounts {
  return {
    likes: Math.max(0, counts.likes + (to === 1 ? 1 : 0) - (from === 1 ? 1 : 0)),
    dislikes: Math.max(0, counts.dislikes + (to === -1 ? 1 : 0) - (from === -1 ? 1 : 0)),
  };
}

export function VoteButtons({
  gameId,
  initialCounts,
}: {
  gameId: string;
  initialCounts: VoteCounts;
}) {
  // The server render has no vote (0); the browser reads the saved one after hydration.
  const vote = useSyncExternalStore(
    subscribe,
    () => readVote(gameId),
    () => 0 as Vote,
  );
  const [counts, setCounts] = useState(initialCounts);
  const [pending, startTransition] = useTransition();

  function choose(target: 1 | -1) {
    if (pending) return;

    const from = vote;
    const to: Vote = from === target ? 0 : target;
    const before = counts;

    // Show the change at once, then replace it with the real numbers (or undo it if saving failed).
    setCounts(applyVote(before, from, to));
    saveVote(gameId, to);

    startTransition(async () => {
      const result = await castVote(gameId, from, to);
      if (result) {
        setCounts(result);
      } else {
        setCounts(before);
        saveVote(gameId, from);
      }
    });
  }

  const total = counts.likes + counts.dislikes;
  const percent = total === 0 ? 0 : Math.round((counts.likes / total) * 100);
  const likes = counts.likes.toLocaleString("en-US");
  const dislikes = counts.dislikes.toLocaleString("en-US");
  const summary =
    total === 0
      ? t("game.noVotes")
      : t("game.voteSummary", {
          percent: String(percent),
          count: total.toLocaleString("en-US"),
        });

  return (
    <div className="flex items-center gap-2 max-sm:gap-0 max-sm:[&>button:nth-child(1)]:rounded-r-none max-sm:[&>button:nth-child(2)]:-ml-px max-sm:[&>button:nth-child(2)]:rounded-l-none">
      <button
        type="button"
        aria-pressed={vote === 1}
        aria-label={t("game.like") + " (" + likes + ")"}
        title={t("game.like")}
        disabled={pending}
        onClick={() => choose(1)}
        className={actionButton}
      >
        <ThumbsUp
          size={18}
          strokeWidth={1.5}
          fill={vote === 1 ? "currentColor" : "none"}
          aria-hidden="true"
        />
        <span className="tabular-nums">{likes}</span>
      </button>
      <button
        type="button"
        aria-pressed={vote === -1}
        aria-label={t("game.dislike") + " (" + dislikes + ")"}
        title={t("game.dislike")}
        disabled={pending}
        onClick={() => choose(-1)}
        className={actionButton}
      >
        <ThumbsDown
          size={18}
          strokeWidth={1.5}
          fill={vote === -1 ? "currentColor" : "none"}
          aria-hidden="true"
        />
        <span className="tabular-nums">{dislikes}</span>
      </button>
      {/* Screen readers only: the vote summary is read out, but no percentage is shown on screen. */}
      <p className="sr-only" aria-live="polite" title={summary}>
        {summary}
      </p>
    </div>
  );
}

function parseRow(row: unknown) {
  const base = parseBase(row);
  if (!base.success) return base;

  const r = row as Record<string, unknown>;
  return {
    success: true as const,
    data: {
      ...base.data,
      playCount: Number(r.play_count) || 0,
      likeCount: Number(r.like_count) || 0,
      dislikeCount: Number(r.dislike_count) || 0,
    },
  };
}

function parseBase(row: unknown) {
  const r = row as Record<string, unknown>;
  const category = r.categories as { slug?: string } | null;

  return gameSchema.safeParse({
    id: r.id,
    slug: r.slug,
    title: r.title,
    description: r.description,
    thumbnailUrl: r.thumbnail_url,
    embedUrl: r.embed_url,
    categorySlug: category?.slug,
    tags: r.tags,
    devices: r.devices,
    provider: r.provider,
    createdAt: r.created_at,
  });
}

// Filters come in as plain strings (null = no filter) so the cache key stays simple.
async function loadGamesPage(
  sort: SortName,
  page: number,
  pageSize: number,
  category: string | null,
  device: string | null,
  tag: string | null,
): Promise<GamesPage> {
  "use cache";
  cacheLife("minutes");

  const supabase = createPublicClient();

  // "!inner" is always used, so the count and the list agree on which games exist.
  // The category filter itself is added only when one is chosen.
  let countQuery = supabase
    .from("games")
    .select("id, categories!inner(slug)", { count: "exact", head: true })
    .eq("is_published", true);

  let listQuery = supabase.from("games").select(GAME_COLUMNS_INNER).eq("is_published", true);

  if (category) {
    countQuery = countQuery.eq("categories.slug", category);
    listQuery = listQuery.eq("categories.slug", category);
  }
  if (device) {
    countQuery = countQuery.contains("devices", [device]);
    listQuery = listQuery.contains("devices", [device]);
  }
  if (tag) {
    countQuery = countQuery.contains("tags", [tag]);
    listQuery = listQuery.contains("tags", [tag]);
  }

  const { count, error: countError } = await countQuery;

  if (countError || count === null) {
    throw new Error(`getGamesPage count failed: ${countError?.message ?? "no count"}`);
  }

  const pageCount = Math.max(1, Math.ceil(count / pageSize));
  if (page > pageCount) return { games: [], total: count, page, pageCount };

  const from = (page - 1) * pageSize;

  const { data, error } = await listQuery
    .order(SORT_COLUMN[sort], { ascending: false })
    .order("id")
    .range(from, from + pageSize - 1);

  if (error) throw new Error(`getGamesPage failed: ${error.message}`);

  return { games: parseRows(data), total: count, page, pageCount };
}

// Candidates: same category, plus games that share a tag.
// Score: 2 for the same category, 1 for each shared tag. Ties go to the most played.
async function loadRelatedGames(
  categorySlug: string,
  excludeSlug: string,
  tags: string[],
  limit: number,
): Promise<Game[]> {
  "use cache";
  cacheLife("minutes");

  const supabase = createPublicClient();

  const sameCategory = await supabase
    .from("games")
    .select(GAME_COLUMNS_INNER)
    .eq("is_published", true)
    .eq("categories.slug", categorySlug)
    .neq("slug", excludeSlug)
    .order("play_count", { ascending: false })
    .order("id")
    .limit(48);

  if (sameCategory.error) {
    throw new Error(`getRelatedGames category failed: ${sameCategory.error.message}`);
  }

  const candidates = parseRows(sameCategory.data);

  if (tags.length > 0) {
    const sharedTags = await supabase
      .from("games")
      .select(GAME_COLUMNS)
      .eq("is_published", true)
      .overlaps("tags", tags)
      .neq("slug", excludeSlug)
      .order("play_count", { ascending: false })
      .order("id")
      .limit(48);

    if (sharedTags.error) {
      throw new Error(`getRelatedGames tags failed: ${sharedTags.error.message}`);
    }

    candidates.push(...parseRows(sharedTags.data));
  }

  const unique = [...new Map(candidates.map((game) => [game.id, game])).values()];

  const score = (game: Game) =>
    (game.categorySlug === categorySlug ? 2 : 0) +
    game.tags.filter((tag) => tags.includes(tag)).length;

  // sort() is stable, so equal scores keep the play_count order.
  return unique.sort((a, b) => score(b) - score(a)).slice(0, limit);
}

  } catch (error) {

    return [];

// ---------- Search ----------

// % _ and \ mean something special inside ilike, so they are escaped.
function escapeLike(value: string) {
  return value.replace(/[\\%_]/g, "\\$&");
}

// Not cached on purpose: visitors can type anything, so the cache would never stop growing.
// Call it inside Suspense. null = the database could not be reached.
export async function searchGames(query: string, limit = LISTING_LIMIT): Promise<Game[] | null> {
  const q = query.trim().toLowerCase();
  if (q.length === 0) return [];

  try {
    const supabase = createPublicClient();

    const byTitle = await supabase
      .from("games")
      .select(GAME_COLUMNS)
      .eq("is_published", true)
      .ilike("title", `%${escapeLike(q)}%`)
      .order("play_count", { ascending: false })
      .order("id")
      .limit(limit);

    if (byTitle.error) throw new Error(`searchGames title failed: ${byTitle.error.message}`);

    const found = parseRows(byTitle.data);

    // Tags are lowercase with dashes ("word-puzzle"), so "word puzzle" is tried as that tag.
    const tag = q.replace(/\s+/g, "-");
    if (SLUG_PATTERN.test(tag)) {
      const byTag = await supabase
        .from("games")
        .select(GAME_COLUMNS)
        .eq("is_published", true)
        .contains("tags", [tag])
        .order("play_count", { ascending: false })
        .order("id")
        .limit(limit);

      if (byTag.error) throw new Error(`searchGames tag failed: ${byTag.error.message}`);
      found.push(...parseRows(byTag.data));
    }

    const unique = [...new Map(found.map((game) => [game.id, game])).values()];

    // 0 = title starts with the search, 1 = title contains it, 2 = tag match only.
    // sort() is stable, so equal ranks keep the play_count order.
    const rank = (game: Game) => {
      const title = game.title.toLowerCase();
      return title.startsWith(q) ? 0 : title.includes(q) ? 1 : 2;
    };

    return unique.sort((a, b) => rank(a) - rank(b)).slice(0, limit);
  } catch (error) {
    console.error(error);
    return null;
  }
}

      .in("slug", unique);

    if (error) throw new Error(`getGamesBySlugs failed: ${error.message}`);

    const bySlug = new Map(parseRows(data).map((game) => [game.slug, game]));
    return unique.flatMap((slug) => {
      const game = bySlug.get(slug);
      return game ? [game] : [];
    });
  } catch (error) {
    console.error(error);
    return null;
  }
}

===== apps\web\lib\vote-count.ts =====
import { createPublicClient } from "@/lib/supabase/public";

export type VoteCounts = { likes: number; dislikes: number };

// Not cached on purpose: the numbers should be fresh. Call it inside Suspense.
// null = game not found or the database could not be reached.
export async function getVoteCounts(gameId: string): Promise<VoteCounts | null> {
  try {
    const { data, error } = await createPublicClient()
      .from("games")
      .select("like_count, dislike_count")
      .eq("id", gameId)
      .eq("is_published", true)
      .maybeSingle();

    if (error) {
      console.error("getVoteCounts failed:", error.message);
      return null;
    }

    return data ? { likes: Number(data.like_count), dislikes: Number(data.dislike_count) } : null;
  } catch (error) {
    console.error("getVoteCounts failed:", error);
    return null;
  }
}
