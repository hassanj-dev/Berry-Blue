===== apps\web\components\layout\icons.tsx =====
import {
  Flame,
  Heart,
  House,
  LayoutGrid,
  Moon,
  Play,
  Search,
  Sparkles,
  Sun,
  ThumbsUp,
  type LucideIcon,
} from "lucide-react";

export type IconName = "home" | "grid" | "search" | "heart" | "sun" | "moon" | "flame" | "sparkles";

const icons: Record<IconName, LucideIcon> = {
  home: House,
  grid: LayoutGrid,
  search: Search,
  heart: Heart,
  sun: Sun,
  moon: Moon,
  flame: Flame,
  sparkles: Sparkles,
};

type IconProps = { name: IconName; size?: number; filled?: boolean };

export function Icon({ name, size = 20, filled = false }: IconProps) {
  const Component = icons[name];
  return (
    <Component
      size={size}
      strokeWidth={1.5}
      fill={filled ? "currentColor" : "none"}
      aria-hidden="true"
    />
  );
}

export function NavIcon({ name }: { name: IconName }) {
  return <Icon name={name} />;
}

export function PlayIcon({ size = 16 }: { size?: number }) {
  return <Play size={size} strokeWidth={1.5} fill="currentColor" aria-hidden="true" />;
}

export function ThumbIcon({ size = 16 }: { size?: number }) {
  return <ThumbsUp size={size} strokeWidth={1.5} fill="currentColor" aria-hidden="true" />;
}

===== apps\web\components\layout\NavLink.tsx =====
"use client";

import Link from "next/link";
import { usePathname } from "next/navigation";
import type { ReactNode } from "react";
import { cn } from "@/lib/cn";

type NavLinkProps = {
  href: string;
  className?: string;
  activeClassName?: string;
  children: ReactNode;
};

export function NavLink({ href, className, activeClassName, children }: NavLinkProps) {
  const pathname = usePathname();
  const active = href === "/" ? pathname === "/" : pathname.startsWith(href);

  return (
    <Link
      href={href}
      aria-current={active ? "page" : undefined}
      className={cn(className, active && activeClassName)}
    >
      {children}
    </Link>
  );
}

===== apps\web\components\layout\Sidebar.tsx =====
import Link from "next/link";
import { Suspense } from "react";
import { Icon } from "@/components/layout/icons";
import { Logo } from "@/components/layout/Logo";
import { NavLink } from "@/components/layout/NavLink";
import { t } from "@/i18n";
import { getCategories } from "@/lib/games";
import { footerNav, sidebarNav } from "@/lib/nav";
import { siteConfig } from "@/lib/site";

const itemClass =
  "flex h-9 items-center gap-2.5 rounded-lg px-3 text-sm text-muted transition-colors hover:bg-surface hover:text-foreground";
const itemActive = "bg-surface text-foreground";

async function SidebarCategories() {
  const categories = await getCategories();
  if (categories.length === 0) return null;

  return (
    <nav aria-labelledby="sidebar-categories">
      <h2 id="sidebar-categories" className="mb-2 px-3 text-xs text-muted">
        {t("nav.categories")}
      </h2>
      <Suspense fallback={null}>
        <ul className="flex flex-col gap-0.5">
          {categories.map((category) => (
            <li key={category.id}>
              <NavLink
                href={"/category/" + category.slug}
                className={itemClass}
                activeClassName={itemActive}
              >
                {category.name}
              </NavLink>
            </li>
          ))}
        </ul>
      </Suspense>
    </nav>
  );
}

export function Sidebar() {
  return (
    <aside className="hidden w-60 shrink-0 border-r border-border lg:block">
      <div className="sticky top-0 flex h-dvh flex-col gap-6 overflow-y-auto overflow-x-hidden scroll-row px-3 py-4">
        <Logo className="h-9 px-3" />

        <nav aria-label={t("nav.main")}>
          <Suspense fallback={null}>
            <ul className="flex flex-col gap-0.5">
              {sidebarNav.map((item) => (
                <li key={item.href}>
                  <NavLink href={item.href} className={itemClass} activeClassName={itemActive}>
                    <Icon name={item.icon} />
                    {t(item.labelKey)}
                  </NavLink>
                </li>
              ))}
            </ul>
          </Suspense>
        </nav>

        <Suspense fallback={null}>
          <SidebarCategories />
        </Suspense>

        <div className="mt-auto px-3">
          <nav aria-label={t("footer.links")}>
            <ul className="flex flex-wrap gap-x-4 gap-y-1 text-xs">
              {footerNav.map((item) => (
                <li key={item.href}>
                  <Link href={item.href} className="text-muted hover:text-foreground">
                    {t(item.labelKey)}
                  </Link>
                </li>
              ))}
            </ul>
          </nav>
          <p className="mt-3 text-xs text-muted">
            {t("footer.copyright", { name: siteConfig.name })}
          </p>
        </div>
      </div>
    </aside>
  );
}

import type { IconName } from "@/components/layout/icons";
import type { MessageKey } from "@/i18n";

export type NavItem = { href: string; labelKey: MessageKey; icon: IconName };

// Phone bottom bar. Order matters.
export const mainNav: NavItem[] = [
  { href: "/", labelKey: "nav.home", icon: "home" },
  { href: "/games", labelKey: "nav.games", icon: "grid" },
  { href: "/search", labelKey: "nav.search", icon: "search" },
  { href: "/favorites", labelKey: "nav.favorites", icon: "heart" },
];

// Desktop sidebar. Search lives in the top bar there.
export const sidebarNav: NavItem[] = mainNav.filter((item) => item.href !== "/search");
